We have two proofs that

      Every list has a reversed list

and then extract reversal functions from them (in Agda/MLTT this is
done via the first projection function pr₁).

The first proof Thm₁ is simple, but the extracted function reverse₁ is
in quadratic time.

The second one Thm₂ is more complicated with the aid of AuxLemma, but
the extracted function reverse₂ is in linear time.

\begin{code}

module ListReversal where

open import Preliminaries

\end{code}

Lists and operations

\begin{code}

infixr 20 _::_
infixl 10 _++_

data List (A : Type) : Type where
 []   : List A
 _::_ : A → List A → List A

[_] : {A : Type} → A → List A
[ a ] = a :: []

_++_ : {A : Type} → List A → List A → List A
[] ++ v = v
(a :: u) ++ v = a :: (u ++ v)

\end{code}

An inductively-defined relation of reversed lists

\begin{code}

data reversed {A : Type} : List A → List A → Type where
 r-nil  : reversed [] []
 r-cons : {u v : List A} {a : A} → reversed u v → reversed (a :: u) (v ++ [ a ])

\end{code}

A simple proof that every list has a reversed list,
whose reversal algorithm is inefficient (quadratic time).

\begin{code}

Thm₁ : {A : Type}
     → ∀ (u : List A) → ∃ \(v : List A) → reversed u v
Thm₁ [] = [] , r-nil
Thm₁ {A} (a :: u) = v ++ [ a ] , r-cons p
 where
  IH : ∃ \(v : List A) → reversed u v
  IH = Thm₁ u
  v : List A
  v = pr₁ IH
  p : reversed u v
  p = pr₂ IH

reverse₁ : {A : Type} → List A → List A
reverse₁ u = pr₁ (Thm₁ u)

\end{code}

The above reverse function can be written as

\begin{code}

reverse¹ : {A : Type} → List A → List A
reverse¹  []      = []
reverse¹ (a :: u) = reverse¹ u ++ [ a ]

\end{code}

Now we have a slightly more complicated proof witha linear reversal algorithm.

\begin{code}

AuxLemma : {A : Type}
    → ∀ (u v : List A)
    → ∃ \(l : List A) → ∃ \(w : List A) → (w ++ v ≡ l) ∧ (reversed u w)
AuxLemma [] v = v , [] , refl , r-nil
AuxLemma {A} (a :: u) v = l , w ++ [ a ] , q , r-cons r
 where
  IH : ∃ \(l : List A) → ∃ \(w : List A) → (w ++ a :: v ≡ l) ∧ (reversed u w)
  IH = AuxLemma u (a :: v)
  l : List A
  l = pr₁ IH
  w : List A
  w = pr₁ (pr₂ IH)
  p : w ++ a :: v ≡ l
  p = pr₁ (pr₂ (pr₂ IH))
  q : w ++ [ a ] ++ v ≡ l
  q = trans (Lemma[u++a++v=u++a::v] w v a) p
  r : reversed u w
  r = pr₂ (pr₂ (pr₂ IH))

Thm₂ : {A : Type}
     → ∀ (u : List A) → ∃ \(v : List A) → reversed u v
Thm₂ {A} u = v , r'
 where
  known : ∃ \(v : List A) → ∃ \(w : List A) → (w ++ [] ≡ v) ∧ (reversed u w)
  known = AuxLemma u []
  v : List A
  v = pr₁ known
  w : List A
  w = pr₁ (pr₂ known)
  p : w ++ [] ≡ v
  p = pr₁ (pr₂ (pr₂ known))
  q : w ≡ v
  q = Lemma[u++[]=v→u=v] p
  r : reversed u w
  r = pr₂ (pr₂ (pr₂ known))
  r' : reversed u v
  r' = transport (reversed u) q r

reverse₂ : {A : Type} → List A → List A
reverse₂ u = pr₁ (Thm₂ u)

\end{code}

The second reverse function is defined with a helper function derived
from AuxLemma as follows

\begin{code}

reverseAux : {A : Type} → List A → List A → List A
reverseAux  []      v = v
reverseAux (a :: u) v = reverseAux u (a :: v)

reverse² : {A : Type} → List A → List A
reverse² u = reverseAux u []

\end{code}
