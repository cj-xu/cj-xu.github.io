Indentity types

\begin{code}

infixr 2 _≡_

data _≡_ {A : Type} : A → A → Type where
 refl : {a : A} → a ≡ a

trans : {A : Type} {a b c : A}
      → a ≡ b → b ≡ c → a ≡ c
trans refl refl = refl

transport : {A : Type} (P : A → Type)
          → {x y : A} → x ≡ y → P x → P y
transport P refl p = p

ap : {A B : Type} {a b : A}
   → (f : A → B) → a ≡ b → f a ≡ f b
ap f refl = refl

\end{code}

(Non-)Dependent products

\begin{code}

infixr 5 _,_

record Σ {A : Type} (B : A → Type) : Type where
 constructor _,_
 field
  pr₁ : A
  pr₂ : B pr₁

open Σ public

_×_ : Type → Type → Type
A × B = Σ \(a : A) → B

\end{code}

Logic

\begin{code}

∃ : {A : Type} → (A → Type) → Type
∃ = Σ

infixr 1 _∧_

_∧_ : Type → Type → Type
_∧_ = _×_

\end{code}
