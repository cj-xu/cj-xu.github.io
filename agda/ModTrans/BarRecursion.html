■ Preliminaries

\begin{code}

data 𝟚 : Set where
 𝟎 𝟏 : 𝟚

min² : 𝟚 → 𝟚 → 𝟚
min² 𝟎 b = 𝟎
min² 𝟏 b = b

Le : ℕ → ℕ → 𝟚
Le  n        0       = 𝟎
Le  0       (succ m) = 𝟏
Le (succ n) (succ m) = Le n m

data _* (A : Set) : Set where
 ⟨⟩  : A *
 _,_ : A → A * → A *

∣_∣ : {A : Set} → A * → ℕ
∣ ⟨⟩ ∣ = 0
∣ _ , u ∣ = succ ∣ u ∣

_✶_ : {A : Set} → A * → A → A *
⟨⟩       ✶ y = y , ⟨⟩
(x , xs) ✶ y = x , (xs ✶ y)

_^ : ℕ * → ℕᴺ
(⟨⟩ ^)        i       = 0
((n , ns) ^)  0       = n
((n , ns) ^) (succ i) = (ns ^) i

≤-dec : (n m : ℕ) → (n > m) + (n ≤ m)
≤-dec  0        m = inr ≤zero
≤-dec (succ n)  0 = inl (≤succ ≤zero)
≤-dec (succ n) (succ m) with ≤-dec n m
≤-dec (succ n) (succ m) | inl r = inl (≤succ r)
≤-dec (succ n) (succ m) | inr r = inr (≤succ r)

_-_ : ℕ → ℕ → ℕ
0      - m      = 0
succ n - 0      = succ n
succ n - succ m = n - m

\end{code}

■ A nucleus for general-bar-recursion functionals

\begin{code}

ℕᵒ : Set
ℕᵒ = (ℕᴺ → ℕ) × (ℕ * → 𝟚) × ((ℕ * → σ) → (ℕ * → (ℕ → σ) → σ) → ℕ * → σ)

V : ℕᵒ → ℕᴺ → ℕ
V = pr₁

S : ℕᵒ → ℕ * → 𝟚
S = pr₁ ∘ pr₂

B : ℕᵒ → (ℕ * → σ) → (ℕ * → (ℕ → σ) → σ) → ℕ * → σ
B = pr₂ ∘ pr₂

η : ℕ → ℕᵒ
η n = (λ α → n) , (λ s → 𝟏) , (λ G H → G)

κ : (ℕ → ℕᵒ) → ℕᵒ → ℕᵒ
κ g x = (λ α → V (g (V x α)) α)
      , (λ s → min² (S x s) (S (g (V x (s ^))) s))
      , (λ G H → B x (λ s → B (g (V x (s ^))) G H s) H)

⟪_⟫ʸ : Ty → Set
⟪ ι ⟫ʸ     = ℕᵒ
⟪ τ ⇾ ρ ⟫ʸ = ⟪ τ ⟫ʸ → ⟪ ρ ⟫ʸ

⟪_⟫ˣ : Cxt → Set
⟪ ε ⟫ˣ = 𝟙
⟪ Γ ₊ ρ ⟫ˣ = ⟪ Γ ⟫ˣ × ⟪ ρ ⟫ʸ

_₍_₎ᵒ : {Γ : Cxt} {ρ : Ty} → ⟪ Γ ⟫ˣ → ∥ ρ ∈ Γ ∥ → ⟪ ρ ⟫ʸ
(_ , a) ₍ zero ₎ᵒ   = a
(γ , _) ₍ succ i ₎ᵒ = γ ₍ i ₎ᵒ

KE : {ρ : Ty} → (ℕ → ⟪ ρ ⟫ʸ) → ℕᵒ → ⟪ ρ ⟫ʸ
KE {ι}     g = κ g
KE {τ ⇾ ρ} g = λ x a → KE (λ z → g z a) x

⟪_⟫ᵐ : {Γ : Cxt} {ρ : Ty} → T Γ ρ → ⟪ Γ ⟫ˣ → ⟪ ρ ⟫ʸ
⟪ Var i ⟫ᵐ γ = γ ₍ i ₎ᵒ
⟪ Lam t ⟫ᵐ γ = λ a → ⟪ t ⟫ᵐ (γ , a)
⟪ f · a ⟫ᵐ γ = ⟪ f ⟫ᵐ γ (⟪ a ⟫ᵐ γ)
⟪ Zero ⟫ᵐ  _ = η 0
⟪ Succ ⟫ᵐ  _ = κ (η ∘ succ)
⟪ Rec ⟫ᵐ   _ = λ a f → KE (rec a (f ∘ η))

_ᵒ : {ρ : Ty} → T ε ρ → ⟪ ρ ⟫ʸ
t ᵒ = ⟪ t ⟫ᵐ ⋆

--
-- Ψ k is a (T-definable) functional of bar recursion
-- for constant Y with value k
-- [Oliva&Steila2018, Lemma 2.1]
--
Ψ : ℕ → (ℕ * → σ) → (ℕ * → (ℕ → σ) → σ) → ℕ * → σ
Ψ k G H s with ≤-dec ∣ s ∣ k
Ψ k G H s | inl |s|>k = G s
Ψ k G H s | inr |s|≤k = ψ G H (succ k - ∣ s ∣) s
 where
  ψ : (ℕ * → σ) → (ℕ * → (ℕ → σ) → σ) → ℕ → ℕ * → σ
  ψ G H  0       = G
  ψ G H (succ n) = λ s → H s (λ x → ψ G H n (s ✶ x))

Ω : ℕᵒ → ℕᵒ
Ω = κ (λ n → (λ α → α n) , (λ s → Le n ∣ s ∣) , Ψ n)

--
-- termination predicate
--
𝑆 : T ε ((ι ⇾ ι) ⇾ ι)
  → ℕ * → 𝟚
𝑆 Y = S ((Y ᵒ) Ω)

--
-- 𝐵 Y is functional of general bar recursion for 𝑆 Y.
--
𝐵 : T ε ((ι ⇾ ι) ⇾ ι)
  → (ℕ * → σ) → (ℕ * → (ℕ → σ) → σ) → ℕ * → σ
𝐵 Y = B ((Y ᵒ) Ω)

\end{code}
